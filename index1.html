function validatePassword(pwd, confirm){

  const rules = {

    len: pwd.length >= 8 && pwd.length <= 10,

    charset: /^[A-Za-z0-9]*$/.test(pwd) && pwd.length > 0,

    letters: (pwd.match(/[A-Za-z]/g) || []).length >= 3,

    numbers: (pwd.match(/[0-9]/g) || []).length >= 3,

    repeat: !/(.)\1{5,}/.test(pwd),

    match: pwd.length > 0 && pwd === confirm

  };

  Object.keys(rules).forEach(k=>{

    const el = document.querySelector(`.pwd-rules .r[data-r="${k}"]`);

    if(el) el.classList.toggle('ok', rules[k]);

  });

  return Object.values(rules).every(Boolean);

}
